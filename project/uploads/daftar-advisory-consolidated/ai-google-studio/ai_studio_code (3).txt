interface CardProps {
  tag?: string;
  title: string;
  description: string;
  items?: string[];
}

export const Card = ({ tag, title, description, items }: CardProps) => (
  <article className="p-6 bg-paper-soft/80 border border-white/45 rounded-daftar shadow-xl transition-transform hover:-translate-y-1">
    {tag && <div className="text-rust font-mono text-[10px] tracking-widest uppercase mb-3">{tag}</div>}
    <h3 className="font-display font-light text-3xl leading-none mb-3 tracking-tight">{title}</h3>
    <p className="text-ink-soft mb-4">{description}</p>
    {items && (
      <ul className="mt-4 border-t border-line">
        {items.map((item, i) => (
          <li key={i} className="py-3 border-b border-line last:border-0 text-sm text-muted">
            {item}
          </li>
        ))}
      </ul>
    )}
  </article>
);
